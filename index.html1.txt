                el.addEventListener('click', function(e) {
                    e.preventDefault();
                    // Копируем номер телефона в буфер обмена
                    const phoneText = '+7 (951) 224-42-77';
                    navigator.clipboard.writeText(phoneText).then(() => {
                        // Показываем уведомление
                        const originalText = el.textContent;
                        el.textContent = '✓ Номер скопирован';
                        el.style.borderColor = '#c9a96e';
                        setTimeout(() => {
                            el.textContent = originalText;
                            el.style.borderColor = '#333';
                        }, 2000);
                    }).catch(() => {
                        // Фолбэк, если clipboard недоступен
                        prompt('Скопируйте номер телефона для MAX:', phoneText);
                    });
                });
            }
        });

        // ---------- КНОПКА "РАССЧИТАТЬ СТОИМОСТЬ" ----------
        const calcBtn = document.getElementById('calcBtn');
        if (calcBtn) {
            calcBtn.addEventListener('click', function(e) {
                e.preventDefault();
                // Скролл к блоку контактов
                document.getElementById('contact').scrollIntoView({ behavior: 'smooth' });
            });
        }

        // ---------- ПЛАВНЫЙ СКРОЛЛ ДЛЯ ЯКОРНЫХ ССЫЛОК ----------
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function(e) {
                const href = this.getAttribute('href');
                if (href === '#') return;
                const target = document.querySelector(href);
                if (target) {
                    e.preventDefault();
                    target.scrollIntoView({ behavior: 'smooth' });
                }
            });
        });

        // ---------- ЭФФЕКТ ПРИ СКРОЛЛЕ (ХЕДЕР) ----------
        window.addEventListener('scroll', function() {
            const header = document.querySelector('header');
            if (window.scrollY > 50) {
                header.style.background = 'rgba(13, 13, 13, 0.95)';
            } else {
                header.style.background = 'rgba(13, 13, 13, 0.85)';
            }
        });
    })();
</script>
